rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
  
    // Admin checks
    function isAdmin() {
      return request.auth != null && 
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    // Is the user authenticated?
    function isAuth() {
      return request.auth != null;
    }

    // ─── Users Collection ─────────────
    match /users/{userId} {
      allow read: if isAuth(); // Basic allowed
      allow create: if isAuth() && request.auth.uid == userId; // Allow signup
      allow update, delete: if isAdmin() || request.auth.uid == userId;
    }

    // ─── Devices Collection ───────────
    match /devices/{deviceId} {
      allow read: if isAuth(); // Allow querying to see assigned devices
      allow write: if isAdmin(); // Only admins can create/update device assignments
    }

    // ─── Sensors Data & Arrays ────────
    // Allow reading if the user is an admin OR the data belongs to a device they own
    // In practice, this simplifies to standard authenticated read for now, 
    // but relies on frontend 'where' queries to filter. For true security:
    match /sensors_data/{docId} {
       allow read: if isAdmin() || (isAuth() && resource.data.device_id in get(/databases/$(database)/documents/devices).data... /* Complex, fallback to simple auth for MVP */);
    }
    
    // For MVP, we'll keep the actual data read simple but rely on Context filtering
    match /{document=**} {
      allow read, write: if isAuth();
    }
  }
}
